\chapter{Stochastic losses on a subset of the simplex}
\label{chap:polytope_lower}

Section 4.2 and Appendix G of \cite{SSKM26}, with the corrected statement and the repaired proof
of \texttt{notes/constants-polytope-lower.md} (area D), which gives every computation in full,
including the variant with Rademacher losses used here. Let $\eta > 0$, $0 < \eps \le 1/3$ with
$100\eta\log(1/\eps) \le 1$, $L = \log(1/\eps)$, $m = \lceil 16L\rceil$, $d = 5m + 2$ (so
$d \le 100L$ and $\eta d \le 1$), $\mu = \sqrt{\eta d}$ and $\tau = \lceil 3/\eta\rceil$. The
coordinates are $X_i$, $Y_{k,i}$ ($k = 1, 2, 3$), $Z_i$ ($i < m$), $P$ and $Q$; Lean indexes them
by $\mathrm{Fin}\,m \times \mathrm{Fin}\,5 \oplus \mathrm{Fin}\,2$ and transports the construction to $\{0, \dots, d - 1\}$ at the
end. For a loss $\ell$, $S(\ell) = \sum_i \ell(Z_i) + \ell(P)$.

\begin{lemma}[{\cite[Lemmas 43, 44]{SSKM26}}]
  \label{lem:polytope_lower_polytope}
  \lean{Schlisselberg2026Hidden.PolytopeLower.polytope,
    Schlisselberg2026Hidden.PolytopeLower.polytope_eq_setOf_constraint,
    Schlisselberg2026Hidden.PolytopeLower.polytope_subset_simplex,
    Schlisselberg2026Hidden.PolytopeLower.uniform_mem_polytope,
    Schlisselberg2026Hidden.PolytopeLower.comparator,
    Schlisselberg2026Hidden.PolytopeLower.comparator_mem_polytope,
    Schlisselberg2026Hidden.PolytopeLower.sum_mul_comparator_eq_zero,
    Schlisselberg2026Hidden.PolytopeLower.bregDiv_comparator_uniform,
    Schlisselberg2026Hidden.PolytopeLower.bregDiv_comparator_uniform_le}
  \leanok
  The paper's polytope $K = \{w \ge 0, Aw = b\}$, with $b = Au_0$, is the set of $w \ge 0$ with
  $w(Y_{1,i}) = w(Y_{2,i}) = w(Y_{3,i})$, $w(Z_i) = w(P)$, $w(X_i) + 3w(Y_{1,i}) + w(Z_i) = 5/d$ and
  $w(P) + w(Q) = 2/d$, which are $4m + 1$ linear equations. It lies in the simplex and contains
  $u_0$ and the comparator $w^*$ with $w^*(X_i) = 5/d$, $w^*(Q) = 2/d$ and $0$ elsewhere, whose loss
  is $0$ for every loss vanishing on the $X$- and $Q$-coordinates. Moreover
  $D_R(w^*, u_0) = (5m\log 5 + 2\log 2)/d \le 9m/d$ for $m \ge 2$.
\end{lemma}

\begin{proof}
  \leanok
  Lean takes the description by coordinates as the definition of $K$ (the paper's Lemma 43 is only
  used through it, erratum E29), and writes the $4m + 1$ equations as a matrix equation. Each block
  $(X_i, Y_{\cdot,i}, Z_i)$ adds up to $5/d$ and $(P, Q)$ to $2/d$. The divergence is a direct
  computation, with $\log 5 \le 2\log 2 + 1/4$ and $\log 2 < 0.7$.
\end{proof}

\begin{lemma}[{\cite[Lemma 26]{SSKM26}}, for the negative entropy on $\R^d$]
  \label{lem:polytope_lower_gap}
  \lean{Schlisselberg2026Hidden.PolytopeLower.objective_sub_objective,
    Schlisselberg2026Hidden.PolytopeLower.entropyGap_nonneg,
    Schlisselberg2026Hidden.PolytopeLower.neg_mul_expRem_le}
  \leanok
  For the objective $\phi(b) = \eta\ip{\ell}{b} + D_R(b, a)$ of a step from $a$, with
  $R(w) = \sum_j w_j\log w_j - w_j$,
  $\phi(b) - \phi(x) = \sum_j(\eta\ell_j + \log x_j - \log a_j)(b_j - x_j)
  + \sum_j(b_j\log b_j - b_j\log x_j - b_j + x_j)$, and each term of the last sum is nonnegative for
  $b_j \ge 0$ and $x_j > 0$. For $s \ge 0$, $c > 0$ and $r \in \R$, with $h(r) = e^{-r} - 1 + r$,
  $-r(s - c) \le s\log s - s\log c - s + c + c\,h(r)$.
\end{lemma}

\begin{proof}
  \leanok
  Algebra; $\log y \ge 1 - 1/y$; the last inequality is the Fenchel inequality for $s\log s - s$
  at the slope $\log c - r$.
\end{proof}

\begin{lemma}[{\cite[Lemma 48]{SSKM26}}, replaced]
  \label{lem:polytope_lower_curve_step}
  \lean{Schlisselberg2026Hidden.PolytopeLower.curve,
    Schlisselberg2026Hidden.PolytopeLower.curve_mem_polytope,
    Schlisselberg2026Hidden.PolytopeLower.level,
    Schlisselberg2026Hidden.PolytopeLower.strictMonoOn_level,
    Schlisselberg2026Hidden.PolytopeLower.exists_level_eq,
    Schlisselberg2026Hidden.PolytopeLower.levelInv,
    Schlisselberg2026Hidden.PolytopeLower.sum_mul_sub_curve,
    Schlisselberg2026Hidden.PolytopeLower.isApproxMirrorDescentStep_curve}
  \leanok
  \uses{lem:polytope_lower_polytope, lem:polytope_lower_gap}
  For $E > 0$ and $q \in (0, 2/d)$, let $W(E, q) \in K$ be the point with $y = (3/d + q)/(E + 3)$ on
  the $Y$-coordinates, $Ey$ on the $X$-coordinates, $2/d - q$ on the $Z$-coordinates and on $P$, and
  $q$ on $Q$; it has positive coordinates. Let
  $f(E, q) = m\log(Ey) + \log q - (m + 1)\log(2/d - q)$, which is strictly increasing in $q$ from
  $(0, 2/d)$ onto $\R$, with inverse $f(E, \cdot)^{-1}$. For a loss with $\ell(X_i) = 0 = \ell(Q)$
  and $\ell(Y_{k,i}) = 1$, and $q, q' \in (0, 2/d)$, the point $W(e^\eta E, q')$ is a
  $q'h(r)$-approximate step of mirror descent on $K$ with the negative entropy from $W(E, q)$, where
  $r = f(e^\eta E, q') - f(E, q) - \eta S(\ell)$. In particular, for
  $q' = f(e^\eta E, \cdot)^{-1}(f(E, q) + \eta S(\ell))$ it is the exact step, and $W(e^\eta E, q)$
  (the frozen step) is $q h(r)$-approximate.
\end{lemma}

\begin{proof}
  \leanok
  \uses{lem:polytope_lower_polytope, lem:polytope_lower_gap}
  By the description of $K$ of Lemma~\ref{lem:polytope_lower_polytope}, the linear part of the gap
  of Lemma~\ref{lem:polytope_lower_gap} between $b \in K$ and $W(e^\eta E, q')$ is
  $r\,(b(Q) - q')$; bound the remaining sum below by its $Q$-term and apply the last inequality of
  Lemma~\ref{lem:polytope_lower_gap}. The paper's modification of the exact step has gaps (erratum
  E33).
\end{proof}

\begin{lemma}[{\cite[Lemmas 10, 46]{SSKM26}}, for this instance]
  \label{lem:polytope_lower_exact_phase}
  \lean{Schlisselberg2026Hidden.PolytopeLower.trajQ_eq_levelInv,
    Schlisselberg2026Hidden.PolytopeLower.curveY_ge}
  \leanok
  \uses{lem:polytope_lower_steps}

\end{lemma}

\begin{proof}
  \leanok

\begin{lemma}[{\cite[Lemma 47]{SSKM26}}, corrected and sharpened]
  \label{lem:polytope_lower_stuck_base}
  \lean{Schlisselberg2026Hidden.PolytopeLower.levelInv_le,
    Schlisselberg2026Hidden.PolytopeLower.two_div_rho_pow_le,
    Schlisselberg2026Hidden.PolytopeLower.mul_le_level_inv}
  \leanok
  \uses{lem:polytope_lower_curve_step, lem:polytope_lower_exact_phase}
  With $\rho = 3E/(E + 3)$, $f(E, \cdot)^{-1}(\gamma) \le (2/d)(2/\rho)^m e^\gamma$ for every
  $\gamma$. If $E \ge e^3$ and $m \ge 16\log(1/\eps)$, then $(2/\rho)^m \le \eps^4$. For
  $0 \le u \le 4$, $\mu \le 1$ and $m \ge 7$, $f(e^u, 1/d) \ge u\mu$.
\end{lemma}

\begin{proof}
  \leanok
  \uses{lem:polytope_lower_curve_step, lem:polytope_lower_exact_phase}
  From $Ey \ge \rho/d$ (Lemma~\ref{lem:polytope_lower_exact_phase}) and $2/d - q \le 2/d$ in $f$;
  then $2/\rho \le 23/30 \le e^{-1/4}$ for $E \ge 20$. The last bound is
  $m\log(4e^u/(e^u + 3)) \ge u\mu$. The paper's Lemma 47 has inverted signs (erratum E32).
\end{proof}

\begin{lemma}[{\cite[Lemmas 45--47]{SSKM26}}, corrected]
  \label{lem:polytope_lower_stuck}
  \lean{Schlisselberg2026Hidden.PolytopeLower.level_exp_mul_sub_level,
    Schlisselberg2026Hidden.PolytopeLower.trajQ_eq_trajQ_of_le,
    Schlisselberg2026Hidden.PolytopeLower.trajQ_le_inv}
  \leanok
  \uses{lem:polytope_lower_steps, lem:polytope_lower_exact_phase, lem:polytope_lower_stuck_base,
    lem:polytope_lower_params}
  Under the constraints of Lemma~\ref{lem:polytope_lower_params}, for losses as in
  Lemma~\ref{lem:polytope_lower_curve_step} with $\abs{S(\ell_t)} \le m + 1$: if $G_k \le k\mu$ for
  $k = \min(n, \tau)$, then $q_n \le 1/d$.
\end{lemma}

\begin{proof}
  \leanok
  \uses{lem:polytope_lower_steps, lem:polytope_lower_exact_phase, lem:polytope_lower_stuck_base,
    lem:polytope_lower_params}
  For $n \le \tau$, $q_n = f_n^{-1}(\eta G_n) \le f_n^{-1}(\eta n\mu) \le 1/d$ by
  Lemmas~\ref{lem:polytope_lower_exact_phase} and~\ref{lem:polytope_lower_stuck_base}. At $\tau$,
  $\eta G_\tau \le 7/2$ and Lemma~\ref{lem:polytope_lower_stuck_base} give
  $q_\tau \le (2e^{7/2}/d)\eps^4$, so $3q_\tau \le \eps$. One round moves $f(\cdot, q)$ by at most
  $m\eta$, so $\abs r \le 1$ and $q_\tau h(r) \le 3q_\tau \le \eps$: every later step is frozen and
  $q_n = q_\tau$.
\end{proof}

\begin{lemma}[{\cite[Lemma 45]{SSKM26}}, replaced]
  \label{lem:polytope_lower_law}
  \lean{Schlisselberg2026Hidden.PolytopeLower.rademacher,
    Schlisselberg2026Hidden.PolytopeLower.lossLaw,
    Schlisselberg2026Hidden.PolytopeLower.ae_forall_isLawLoss,
    Schlisselberg2026Hidden.PolytopeLower.ae_mem_Icc,
    Schlisselberg2026Hidden.PolytopeLower.measurePreserving_flipZ,
    Schlisselberg2026Hidden.PolytopeLower.half_le_measureReal_sum_le_zero,
    Schlisselberg2026Hidden.PolytopeLower.integral_mul_eq_zero}
  \leanok
  The law of the losses has independent coordinates: $0$ on the $X$- and $Q$-coordinates, $1$ on
  the $Y$-coordinates, $\mu$ on $P$, and Rademacher signs on the $Z$-coordinates; the losses lie in
  $[-1, 1]^d$. For i.i.d.\ losses of that law, flipping the signs of the $Z$-coordinates of a finite
  set of rounds preserves the law of the sequence. Hence the median event
  $\{\sum_{s<n}\sum_i\ell_s(Z_i) \le 0\} = \{G_n \le n\mu\}$ has probability at least $1/2$, and
  $\E[\ell_t(Z_i)\,\varphi] = 0$ for every $\varphi$ invariant under the flip of round $t$.
\end{lemma}

\begin{proof}
  \leanok
  The flip maps the event to $\{G_n \ge n\mu\}$, and the two cover the space. The paper's event
  $\{G \le 0\}$, for Gaussian losses, has probability about $5\cdot10^{-5}$ (erratum E30).
\end{proof}

\begin{lemma}[{\cite[Theorem 49]{SSKM26}}, corrected]
  \label{lem:polytope_lower_expected_regret}
  \lean{Schlisselberg2026Hidden.PolytopeLower.integral_regret_eq,
    Schlisselberg2026Hidden.PolytopeLower.le_integral_regret}
  \leanok
  \uses{lem:polytope_lower_polytope, lem:polytope_lower_steps, lem:polytope_lower_exact_phase,
    lem:polytope_lower_stuck, lem:polytope_lower_law}
  For i.i.d.\ losses of the law of Lemma~\ref{lem:polytope_lower_law},
  $\E[\mathrm{Regret}_T(w^*)] = \sum_{t<T}(3m\,\E[y_t] + \mu\,\E[2/d - q_t])$, which is at least
  $T\mu/(2d)$, and at least $3m/(2d\eta) + T\mu/(2d)$ if $\eta T \ge 1$.
\end{lemma}

\begin{proof}
  \leanok
  \uses{lem:polytope_lower_polytope, lem:polytope_lower_steps, lem:polytope_lower_exact_phase,
    lem:polytope_lower_stuck, lem:polytope_lower_law}
  The comparator has zero loss. The play of round $t$ depends on the past losses only, so it is
  invariant under the flip of round $t$ and the $Z$-terms of $\ip{\ell_t}{w_t}$ have mean $0$
  (Lemma~\ref{lem:polytope_lower_law}). On the median event of $\min(t, \tau)$ rounds, $q_t \le 1/d$
  (Lemma~\ref{lem:polytope_lower_stuck}), so $\E[2/d - q_t] \ge 1/(2d)$ by Markov's inequality. By
  Lemma~\ref{lem:polytope_lower_exact_phase}, $y_t \ge (3/d)/(e^{\eta t} + 3) \ge 1/(2d)$ when
  $\eta t \le 1$; if $\eta T \ge 1$, at least $1/\eta$ rounds $t < T$ have $\eta t \le 1$, so
  $\sum_{t<T}\E[y_t] \ge 1/(2d\eta)$. The paper computes the
  expectation on an event that depends on the losses of the same rounds (erratum E34).
\end{proof}

\begin{theorem}[{\cite[Theorem 8]{SSKM26}}, corrected]
  \label{thm:lb_nonsimplex}
  \lean{Schlisselberg2026Hidden.exists_approxMirrorDescent_le_integral_regret_polytope}
  \leanok
  \uses{lem:polytope_lower_params, lem:polytope_lower_polytope, lem:polytope_lower_steps,
    lem:polytope_lower_law, lem:polytope_lower_expected_regret}
  With $\eta, \eps$ as above, there are $d \le 100\log(1/\eps)$, a polytope $K \subseteq \Delta_d$ in
  standard form containing $u_0$, a law of losses in $[-1, 1]^d$, a trajectory depending measurably
  on the past losses only, almost surely $\eps$-approximate from $u_0$ with positive coordinates,
  and $w^* \in K$, such that for i.i.d.\ losses of that law and every $T$,
  $\E[\mathrm{Regret}_T(w^*)] \ge T\sqrt{\eta/\log(1/\eps)}/20$, and
  $\E[\mathrm{Regret}_T(w^*)] \ge D_R(w^*, u_0)/(6\eta) + T\sqrt{\eta/\log(1/\eps)}/20$ if $\eta T \ge 1$.
\end{theorem}

\begin{proof}
  \leanok
  \uses{lem:polytope_lower_params, lem:polytope_lower_polytope, lem:polytope_lower_steps,
    lem:polytope_lower_law, lem:polytope_lower_expected_regret}
  Transport the construction to $\{0, \dots, d - 1\}$ along an enumeration of the coordinates: the
  objectives of the steps, the regret and $D_R(\cdot, u_0)$ are invariant. By
  Lemma~\ref{lem:polytope_lower_expected_regret}, the expected regret is at least
  $T\mu/(2d) = T\sqrt{\eta/d}/2 \ge T\sqrt{\eta/\log(1/\eps)}/20$
  (Lemma~\ref{lem:polytope_lower_params}), and if $\eta T \ge 1$ the $Y$-coordinates add
  $3m/(2d\eta) \ge D_R(w^*, u_0)/(6\eta)$ (Lemma~\ref{lem:polytope_lower_polytope}).
\end{proof}
